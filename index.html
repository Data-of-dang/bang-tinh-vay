<script>

    // 👇 DÁN LINK CỦA MÀY VÀO ĐÂY (Giữa 2 dấu ngoặc kép) 👇

    const API_URL = "LINK_API_CUA_MAY";

    // 👆 DÁN LINK CỦA MÀY VÀO ĐÂY 👆


    let bankData = [];

    let promoData = [];


    function formatCurrency(input) {

        let value = input.value.replace(/,/g, '');

        if (!isNaN(value) && value !== "") {

            input.value = Number(value).toLocaleString('en-US');

        }

    }


    function parseValue(id) {

        return Number(document.getElementById(id).value.replace(/,/g, ''));

    }


    // Hàm an toàn để chuyển đổi số từ Sheet (xử lý cả dấu phẩy và chấm)

    function safeParseNumber(val) {

        if (!val) return 0;

        return Number(val.toString().replace(',', '.'));

    }


    async function fetchData() {

        if (API_URL === "LINK_API_CUA_MAY" || API_URL === "") {

            document.getElementById('api-error').style.display = 'block';

            document.getElementById('promo-container').innerHTML = '';

            return;

        }


        try {

            const response = await fetch(API_URL);

            const data = await response.json();

            

            // Lọc bỏ các dòng rỗng

            bankData = data.banks.filter(b => b.Ten_Ngan_Hang);

            promoData = data.promos.filter(p => p.Ten_Goi);

            

            renderOptions();

        } catch (error) {

            console.error("Lỗi:", error);

            document.getElementById('promo-container').innerHTML = '<label style="color:red">Lỗi kết nối dữ liệu!</label>';

        }

    }


    function renderOptions() {

        // Đổ data ưu đãi

        let promoHTML = '';

        promoData.forEach((promo, index) => {

            promoHTML += `<label><input type="checkbox" class="promo-checkbox" value="${index}"> ${promo.Ten_Goi}</label>`;

        });

        document.getElementById('promo-container').innerHTML = promoHTML || '<label>Không có ưu đãi nào</label>';


        // Đổ data ngân hàng

        let bankHTML = '';

        bankData.forEach((bank, index) => {

            let rateDisplay = safeParseNumber(bank.Lai_Uu_Dai);

            bankHTML += `<option value="${index}">${bank.Ten_Ngan_Hang} (Ưu đãi: ${rateDisplay}%)</option>`;

        });

        document.getElementById('bank-select').innerHTML = bankHTML;

    }


    function calculateLoan() {

        if(bankData.length === 0) return; // Tránh lỗi click khi chưa load xong


        document.getElementById('loading').style.display = 'block';

        document.getElementById('result-section').style.display = 'none';


        setTimeout(() => { 

            let propertyValue = parseValue('property-value');

            let loanRatio = Number(document.querySelector('input[name="loan-ratio"]:checked').value) / 100;

            let loanTermYears = parseValue('loan-term');

            

            let loanAmount = propertyValue * loanRatio;

            let totalMonths = loanTermYears * 12;


            // Tính ưu đãi

            let selectedPromoIndexes = Array.from(document.querySelectorAll('.promo-checkbox:checked')).map(cb => cb.value);

            

            let graceMonths = 0;

            let graceType = "KHONG_AN_HAN";

            let totalCashback = 0;


            selectedPromoIndexes.forEach(idx => {

                let p = promoData[idx];

                if(p.Loai_An_Han === "AN_HAN_GOC_LAI" || p.Loai_An_Han === "AN_HAN_GOC") {

                    graceMonths = Math.max(graceMonths, safeParseNumber(p.So_Thang_An_Han));

                    graceType = p.Loai_An_Han;

                }

                totalCashback += safeParseNumber(p.Qua_Tang_Tien_Mat);

            });


            // Vốn tự có

            let ownCapital = propertyValue - loanAmount - totalCashback;

            document.getElementById('own-capital-display').innerText = ownCapital.toLocaleString('vi-VN') + " đ";


            // Thông số ngân hàng

            let selectedBank = bankData[document.getElementById('bank-select').value];

            

            let promoRate = safeParseNumber(selectedBank.Lai_Uu_Dai) / 100 / 12; 

            let floatRate = safeParseNumber(selectedBank.Lai_Tha_Noi) / 100 / 12;

            let bankPromoMonths = safeParseNumber(selectedBank.So_Thang_Uu_Dai);


            let scheduleHTML = '';

            let currentBalance = loanAmount;

            let payingPrincipalMonths = totalMonths - graceMonths;

            

            // Xử lý nếu thời gian ân hạn lớn hơn hoặc bằng tổng thời gian vay (dù hiếm xảy ra)

            let monthlyPrincipal = payingPrincipalMonths > 0 ? (loanAmount / payingPrincipalMonths) : 0;


            // In ra 24 tháng đầu

            let displayMonths = Math.min(24, totalMonths); 


            for(let i = 1; i <= displayMonths; i++) {

                let interestThisMonth = 0;

                let principalThisMonth = 0;


                let currentRate = (i <= bankPromoMonths) ? promoRate : floatRate;

                interestThisMonth = currentBalance * currentRate;


                if (i <= graceMonths) {

                    principalThisMonth = 0;

                    if (graceType === "AN_HAN_GOC_LAI") interestThisMonth = 0;

                } else {

                    principalThisMonth = monthlyPrincipal;

                }


                let totalPayment = principalThisMonth + interestThisMonth;


                scheduleHTML += `

                    <tr>

                        <td>${i}</td>

                        <td>${(principalThisMonth/1000000).toFixed(1)}</td>

                        <td>${(interestThisMonth/1000000).toFixed(1)}</td>

                        <td><strong>${(totalPayment/1000000).toFixed(1)}</strong></td>

                    </tr>

                `;


                currentBalance -= principalThisMonth;

            }


            document.querySelector('#schedule-table tbody').innerHTML = scheduleHTML;

            document.getElementById('loading').style.display = 'none';

            document.getElementById('result-section').style.display = 'block';


        }, 400); // Đợi 0.4s tạo cảm giác hệ thống đang tính toán

    }


    fetchData();

</script>
